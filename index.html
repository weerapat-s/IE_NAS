// เปลี่ยนส่วนการอัปโหลดรูปภาพมาใช้ตัวนี้แทนครับ const uploadBtn = document.getElementById('upload-btn'); const photoInput = document.getElementById('photo-input'); const IMGBB_API_KEY = "ใส่_API_KEY_ที่คุณก๊อปมาตรงนี้ครับ"; uploadBtn.onclick = async () => { const file = photoInput.files[0]; if (!file) return alert("เลือกรูปก่อนครับคุณวีระภัทร!"); uploadBtn.disabled = true; uploadBtn.innerText = "กำลังอัปโหลด..."; // ใช้ FormData เพื่อเตรียมส่งไฟล์ const formData = new FormData(); formData.append("image", file); try { // 🚀 ส่งรูปไปที่ ImgBB (ฟรีและไม่ต้องผูกบัตร) const response = await fetch(`https://api.imgbb.com/1/upload?key=${IMGBB_API_KEY}`, { method: "POST", body: formData }); const result = await response.json(); if (result.success) { const imageUrl = result.data.url; // ✅ นำลิงก์รูปไปโชว์ใน Gallery addImgToGallery(imageUrl); // 💡 ทริค: เราสามารถเก็บ imageUrl นี้ลงใน Firebase Database (ที่ฟรี) // เพื่อให้เวลาเปิดเว็บใหม่รูปยังคงอยู่นะครับ alert("อัปโหลดสำเร็จ! รูปไปอยู่ที่ ImgBB แล้วครับ"); } } catch (error) { console.error(error); alert("อัปโหลดไม่ผ่าน ลองเช็คเน็ตดูครับ!"); } finally { uploadBtn.disabled = false; uploadBtn.innerText = "Upload Photo"; } };
